# ========================================== # Project: S-Localhost # Developed by: Sgdev (Shubhomoy) # Core: Python + CustomTkinter (C++ Nuitka) # ========================================== import http.server import socketserver import threading import socket import customtkinter as ctk import os class SLocalhost(ctk.CTk): def __init__(self): super().__init__() self.title("S-Localhost - Portable Server") self.geometry("600x400") self.server_thread = None self.is_running = False # UI Setup self.header = ctk.CTkLabel(self, text="S-Localhost Server", font=("Arial", 24, "bold")) self.header.pack(pady=20) self.ip_label = ctk.CTkLabel(self, text=f"Local IP: {self.get_local_ip()}:8000") self.ip_label.pack(pady=10) self.start_btn = ctk.CTkButton(self, text="Start Server", command=self.start_server) self.start_btn.pack(pady=20) def get_local_ip(self): try: s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM) s.connect(("8.8.8.8", 80)) ip = s.getsockname()[0] s.close() return ip except Exception: return "127.0.0.1" def run_server(self): PORT = 8000 Handler = http.server.SimpleHTTPRequestHandler with socketserver.TCPServer(("", PORT), Handler) as httpd: print(f"Serving at port {PORT}") httpd.serve_forever() def start_server(self): if not self.is_running: self.server_thread = threading.Thread(target=self.run_server, daemon=True) self.server_thread.start() self.start_btn.configure(text="Server Running (Port 8000)", fg_color="green") self.is_running = True if __name__ == "__main__": app = SLocalhost() app.mainloop()
